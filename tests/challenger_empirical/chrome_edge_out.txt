    function parseDl(dl) {
      const nodes = [];
      const children = Array.from(dl.children);
      for (let i = 0; i < children.length; i++) {
        const el = children[i];
        if (el.tagName.toUpperCase() === 'DT') {
          const h3 = el.querySelector(':scope > h3');
          const a = el.querySelector(':scope > a');
          if (h3) {
            const folderTitle = h3.textContent || 'Untitled Folder';
            const nextEl = children[i + 1];
            let subDl = el.querySelector(':scope > dl');
            if (!subDl && nextEl && nextEl.tagName.toUpperCase() === 'DL') {
              subDl = nextEl;
              i++;
            }
            nodes.push({
              title: folderTitle,
              children: subDl ? parseDl(subDl) : []
            });
          } else if (a) {
            nodes.push({ title: a.textContent || '', url: a.getAttribute('HREF') || '' });
          }
        }
      }
      return nodes;
    }
